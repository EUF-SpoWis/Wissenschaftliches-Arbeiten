## Das Paradigma der Qualitativen Inhaltsanalyse

<!-- class="animate__animated animate__fadeIn" -->
Die **Qualitative Inhaltsanalyse** gehört zu den am häufigsten eingesetzten Auswertungsverfahren in der Sportwissenschaft. Ihr Hauptziel ist die **Komplexitätsreduktion** bei gleichzeitiger Erhaltung des inhaltlichen Sinns.

<h3 style="color: #1a9f8d; margin-top: 0;">Definition nach Kuckartz & Rädiker (2022)</h3>

> „Unter **Qualitativer Inhaltsanalyse** wird die systematische und methodisch kontrollierte wissenschaftliche Analyse von Texten, Bildern, Filmen und anderen Inhalten von Kommunikation verstanden. Es werden nicht nur manifeste, sondern auch latente Inhalte analysiert. Im Zentrum steht die **Kategorie**, mit der das gesamte für die Forschungsfrage bedeutsame Material codiert wird.“

## Grundbegriffe & Verfahren im Überblick

### Das methodische Handwerkszeug

| Begriff | Definition & Bedeutung im Forschungsprozess |
| :--- | :--- |
| **Kategorie** | Ergebnis der Klassifizierung von Einheiten (Aussagen, Handlungen, Ideen). Kann sich auf *Fakten*, *Themen*, *Evaluationen* oder *Analysen* beziehen. |
| **Kategoriensystem** | Gesamtheit aller definierten Kategorien, geordnet als hierarchischer Baum, Liste oder Netzwerk. |
| **Codieren** | Der Prozess des Analysierens, Benennens, Zuordnens und theoretischen Einordnens von Textstellen. |
| **Kodiereinheit** | Ein konkretes Textsegment, das codiert wird. Muss so gewählt sein, dass es auch ausserhalb des Kontextes verständlich bleibt. |
| **Memo** | Schriftliches Festhalten von Gedanken, Ideen, Vermutungen und Interpretationsansätzen während der Analyse (wie digitale Post-its). |

---

### dominierende qualitative Inhaltsanalysen

* **Mayring:** Sehr regelgeleiteter, klassischer Ansatz. Unterscheidet z. B. zwischen *zusammenfassender*, *explizierender* und *strukturierender* Inhaltsanalyse.
* **Kuckartz:** Fokussiert auf die *thematisch strukturierende qualitative Inhaltsanalyse*. Sehr praxistauglich und optimal abgestimmt auf CAQDAS-Software (z. B. MAXQDA).
* **Braun & Clarke (Thematic Analysis):** Sehr flexibler, reflexiver Ansatz zur Erfassung und Musterbildung von Themen im Datenmaterial (vor allem im angloamerikanischen Raum verbreitet).

function newBlock(title){
  cur = {title: title, items: []};
  BLOCKS.push(cur);

function parse(){
  var lines = RAW.split('\n');
  for(var i = 0; i < lines.length; i++){
    var l = lines[i].trim();
    if(!l) continue;
    var m = l.match(/^\[(Einstiegsphase|Themenblock|Ausklang)[^\]]*\]$/);
    if(m){ newBlock(l.slice(1, -1)); continue; }
    if(!cur) newBlock('Beginn');
    var n = l.match(/^S([12])\s*:\s*(.*)$/);
    if(n){
      nr++;
      cur.items.push({typ: 'turn', s: n[1], text: n[2], nr: nr});
    } else if(l.charAt(0) === '[' && l.charAt(l.length - 1) === ']'){
      cur.items.push({typ: 'note', text: l.slice(1, -1)});
    } else if(cur.items.length){
      var last = cur.items[cur.items.length - 1];
      if(last.typ === 'turn'){ last.text += ' ' + l; }
    }
  }
}
function esc(t){
  return String(t).split('&').join('&amp;').split('<').join('&lt;').split('>').join('&gt;');
}
function hl(text, query){
  if(!query) return esc(text);
  var low = text.toLowerCase();
  var ql = query.toLowerCase();
  var out = '';
  var i = 0;
  var j = low.indexOf(ql, i);
  while(j > -1){
    out += esc(text.slice(i, j)) + '<mark>' + esc(text.slice(j, j + query.length)) + '</mark>';
    i = j + query.length;
    j = low.indexOf(ql, i);
  }
  return out + esc(text.slice(i));
}
function shortTitle(t){

  if(m) return m[1] + ' · ' + m[2];
  if(t.indexOf('Einstieg') === 0) return 'Einstieg';
  return t;
}

### Schritt 2: Entwickeln von thematischen Hauptkategorien

- **Herkunft der Hauptkategorien:** Werden in der Regel **deduktiv** (theoriegeleitet) direkt aus den Forschungsfragen abgeleitet.
- **Probelauf (Pilotierung):** Testen des vorläufigen Kategoriensystems an ca. 10–25 % des Datenmaterials.

> **Güte- und Qualitätskriterien für Hauptkategorien:**
> * **Forschungsfragenbezug:** Enge Anbindung an das wissenschaftliche Erkenntnisinteresse.
> * **Erschöpfung:** Für jeden bedeutsamen Aspekt im Text existiert eine Kategorie.
> * **Trennschärfe:** Eindeutige Abgrenzbarkeit zwischen den einzelnen Kategorien.
> * **Innere Kohärenz (Gestalt):** Kategorien bilden ein schlüssiges Gesamtbild.
> * **Verständlichkeit:** Klare Benennungen für Forschende und Leser:innen.

**Hier müsste dann noch die Entwicklung der deduktiven Hauptkategorien als Beispielhandlung dargestellt werden (***keine interaktive Übung an dieser Stelle***)**.

var segs = [];
var marks = {};
var subsel = {};
var phase = 1;
var tool = 'k1';
var SYM = {
  ok: '<span class="qc-ok">✓</span>',
  near: '<span class="qc-near">≈</span>',
  sub: '<span class="qc-half">◐</span>',
  bad: '<span class="qc-bad">✗</span>'
};
function esc(t){
  return String(t).split('&').join('&amp;').split('<').join('&lt;').split('>').join('&gt;');
}
function prep(){
  for(var a = 0; a < AUS.length; a++){
    var us = AUS[a].units;
    for(var i = 0; i < us.length; i++){
      var d = us[i];
      var o = {idx: segs.length, n: i + 1, lines: d[0], k: d[1] || null, s: d[2] || null, e: d[3] || '', alt: d[4] || [], a: a};
      us[i] = o;
      segs.push(o);
    }
  }
}
function altHaupt(o, m){
  for(var i = 0; i < o.alt.length; i++){ if(o.alt[i].split(':')[0] === m) return true; }
  return false;
}
function altHit(o, m, sb){
  for(var i = 0; i < o.alt.length; i++){ if(o.alt[i] === m + ':' + sb) return true; }
  return false;
}
function res(o){
  var m = marks[o.idx] || null;
  var sb = subsel[o.idx] || null;
  if(m === o.k){
    if(!SUBS_AN || !o.k || sb === o.s) return 'ok';
    return altHit(o, m, sb) ? 'near' : 'sub';
  }
  if(m && altHaupt(o, m)){
    if(!SUBS_AN || !sb || altHit(o, m, sb)) return 'near';
    return 'sub';
  }
  return 'bad';
}
function lbl(k, s){
  return KAT[k].kurz + (SUBS_AN && s ? ' · ' + KAT[k].subs[s] : '');
}
function myLabel(o){
  var m = marks[o.idx] || null;
  return m ? lbl(m, subsel[o.idx]) : 'nicht markiert';
}
function solLabel(o){
  return o.k ? lbl(o.k, o.s) : 'unmarkiert';
}
function altLabel(o){
  var r = [];
  for(var i = 0; i < o.alt.length; i++){
    var p = o.alt[i].split(':');
    r.push(lbl(p[0], p[1]));
  }
  return r.join(' oder ');
}
function chip(k, text){
  var st = k ? ' style="background:' + KAT[k].farbe + ';border-color:' + KAT[k].rand + '"' : '';
  return '<span class="qc-chip"' + st + '>' + esc(text) + '</span>';
}
function selectHtml(i, m){
  var subs = KAT[m].subs;
  var h = '<select class="qc-sel" data-sub="' + i + '"><option value="">Subkategorie wählen ...</option>';
  for(var key in subs){
    h += '<option value="' + key + '"' + (subsel[i] === key ? ' selected' : '') + '>' + esc(subs[key]) + '</option>';
  }
  return h + '</select>';
}
function unitHtml(o){
  var m = marks[o.idx] || null;
  var st = '';
  var tag = '<span class="qc-nr">Einheit ' + o.n + '</span>';
  if(m){
    var c = KAT[m];
    st = ' style="background:' + c.farbe + ';border:1px solid ' + c.rand + ';border-left:6px solid ' + c.rand + '"';
    tag += '<span class="qc-tag">' + c.kurz + '</span>';
  }
  var h = '<div class="qc-unit' + (phase === 1 ? '' : ' locked') + '" data-seg="' + o.idx + '"' + st + '>' + tag;
  for(var i = 0; i < o.lines.length; i++){
    h += '<div class="qc-ln ' + (o.lines[i][0] === 'S1' ? 's1' : 's2') + '"><b>' + o.lines[i][0] + ':</b> ' + esc(o.lines[i][1]) + '</div>';
  }
  return h + '</div>';
}
function extras(o){
  var m = marks[o.idx] || null;
  if(phase === 2 && m && SUBS_AN){ return '<div class="qc-x">Subkategorie: ' + selectHtml(o.idx, m) + '</div>'; }
  if(phase === 3 && (m || o.k)){ return '<div class="qc-x">' + SYM[res(o)] + '</div>'; }
  return '';
}
function wrap(o){
  return '<div data-w="' + o.idx + '">' + unitHtml(o) + extras(o) + '</div>';
}
function fbHtml(a){
  var h = '';
  for(var i = 0; i < segs.length; i++){
    var o = segs[i];
    if(o.a !== a) continue;
    var r = res(o);
    if(r === 'ok' && !o.e) continue;
    h += '<div class="qc-fbi"><b>Einheit ' + o.n + '</b> ' + SYM[r] +
         '<div class="qc-small">Deine Codierung: ' + chip(marks[o.idx] || null, myLabel(o)) +
         ' Musterlösung: ' + chip(o.k, solLabel(o)) +
         (o.alt.length ? '<br>Auch vertretbar: <b>' + esc(altLabel(o)) + '</b>' : '') + '</div>' +
         (o.e ? '<div>' + esc(o.e) + '</div>' : '') + '</div>';
  }
  return h ? '<div class="qc-fb"><b>Musterlösung zu diesem Ausschnitt</b>' + h + '</div>' : '';
}
function scoreHtml(){
  var n = segs.length;
  var hk = 0;
  var hn = 0;
  var sm = 0;
  var sk = 0;
  for(var i = 0; i < n; i++){
    var o = segs[i];
    var m = marks[o.idx] || null;
    if(m === o.k){
      hk++;
      if(SUBS_AN && o.k){ sm++; if(subsel[o.idx] === o.s){ sk++; } }
    } else if(m && altHaupt(o, m)){ hn++; }
  }
  var h = '<div class="qc-score"><b>Hauptkategorie:</b> ' + hk + ' von ' + n + ' Sinneinheiten stimmen mit der Musterlösung überein';
  if(hn > 0){ h += ' (dazu ' + hn + ' vertretbare Alternativen ≈)'; }
  h += '.';
  if(SUBS_AN){ h += '<br><b>Subkategorie:</b> ' + sk + ' von ' + sm + ' Einheiten (gezählt nur dort, wo die Hauptkategorie stimmt und die Musterlösung eine Subkategorie hat).'; }
  return h + '<br>Die Musterlösung ist ein Vorschlag und Anlass zum Vergleich im Plenum.</div>';
}
function stepText(){
  if(phase === 1){ return '<b>Schritt 1' + (SUBS_AN ? ' von 2' : '') + ':</b> Markiere die Sinneinheiten mit dem passenden Stift.'; }
  if(phase === 2){ return '<b>Schritt 2 von 2:</b> Wähle für jede markierte Einheit die passende Subkategorie.'; }
  return '<b>Musterlösung:</b> ✓ stimmt &middot; ≈ vertretbare Alternative (Grenzfall) &middot; ◐ Hauptkategorie stimmt, Subkategorie weicht ab &middot; ✗ weicht ab';
}

function render(){
  var root = document.getElementById('qc-widget');
  if(!root) return;
  var h = '<div class="qc-card"><div class="qc-top">';

}
function onClick(e){
  var el = e.target.closest('[data-tool],[data-seg],[data-act]');
  if(!el) return;
  var root = document.getElementById('qc-widget');
  if(el.hasAttribute('data-tool')){
    tool = el.getAttribute('data-tool');
    var bs = root.querySelectorAll('[data-tool]');
    for(var b = 0; b < bs.length; b++){ bs[b].classList.toggle('active', bs[b].getAttribute('data-tool') === tool); }
  } else if(el.hasAttribute('data-seg')){
    if(phase !== 1) return;
    var i = parseInt(el.getAttribute('data-seg'), 10);
    var old = marks[i] || null;
    marks[i] = (tool === 'none' || old === tool) ? null : tool;
    delete subsel[i];
    root.querySelector('[data-w="' + i + '"]').innerHTML = unitHtml(segs[i]) + extras(segs[i]);
  } else {
    var act = el.getAttribute('data-act');
    if(act === 'next'){ phase = SUBS_AN ? 2 : 3; }
    if(act === 'back'){ phase = 1; }
    if(act === 'check'){ phase = 3; }
    if(act === 'reset'){ marks = {}; subsel = {}; phase = 1; }
    render();
    root.scrollIntoView();
  }
}
function onChange(e){
  if(!e.target.hasAttribute('data-sub')) return;
  var i = parseInt(e.target.getAttribute('data-sub'), 10);
  if(e.target.value){ subsel[i] = e.target.value; } else { delete subsel[i]; }
}
function init(tries){
  var root = document.getElementById('qc-widget');
  if(!root){
    if(tries > 0){ setTimeout(function(){ init(tries - 1); }, 250); }
    return;
  }
  if(root.getAttribute('data-ready')) return;
  root.setAttribute('data-ready', '1');
  prep();
  addCss();
  render();
  root.addEventListener('click', onClick);
  root.addEventListener('change', onChange);
}

  return 'bad';
}
function lbl(k, s){
  return KAT[k].kurz + (SUBS_AN && s ? ' · ' + KAT[k].subs[s] : '');
}
function myLabel(o){
  var m = marks[o.idx] || null;
  return m ? lbl(m, subsel[o.idx]) : 'nicht markiert';
}
function solLabel(o){
  return o.k ? lbl(o.k, o.s) : 'unmarkiert';
}
function altLabel(o){
  var r = [];
  for(var i = 0; i < o.alt.length; i++){
    var p = o.alt[i].split(':');
    r.push(lbl(p[0], p[1]));
  }
  return r.join(' oder ');
}
function chip(k, text){
  var st = k ? ' style="background:' + KAT[k].farbe + ';border-color:' + KAT[k].rand + '"' : '';
  return '<span class="qc-chip"' + st + '>' + esc(text) + '</span>';
}
function selectHtml(i, m){
  var subs = KAT[m].subs;
  var h = '<select class="qc-sel" data-sub="' + i + '"><option value="">Subkategorie wählen ...</option>';
  for(var key in subs){
    h += '<option value="' + key + '"' + (subsel[i] === key ? ' selected' : '') + '>' + esc(subs[key]) + '</option>';
  }
  return h + '</select>';
}
function unitHtml(o){
  var m = marks[o.idx] || null;
  var st = '';
  var tag = '<span class="qc-nr">Einheit ' + o.n + '</span>';
  if(m){
    var c = KAT[m];
    st = ' style="background:' + c.farbe + ';border:1px solid ' + c.rand + ';border-left:6px solid ' + c.rand + '"';
    tag += '<span class="qc-tag">' + c.kurz + '</span>';
  }
  var h = '<div class="qc-unit' + (phase === 1 ? '' : ' locked') + '" data-seg="' + o.idx + '"' + st + '>' + tag;
  for(var i = 0; i < o.lines.length; i++){
    h += '<div class="qc-ln ' + (o.lines[i][0] === 'S1' ? 's1' : 's2') + '"><b>' + o.lines[i][0] + ':</b> ' + esc(o.lines[i][1]) + '</div>';
  }
  return h + '</div>';
}
function extras(o){
  var m = marks[o.idx] || null;
  if(phase === 2 && m && SUBS_AN){ return '<div class="qc-x">Subkategorie: ' + selectHtml(o.idx, m) + '</div>'; }
  if(phase === 3 && (m || o.k)){ return '<div class="qc-x">' + SYM[res(o)] + '</div>'; }
  return '';
}
function wrap(o){
  return '<div data-w="' + o.idx + '">' + unitHtml(o) + extras(o) + '</div>';
}
function fbHtml(a){
  var h = '';
  for(var i = 0; i < segs.length; i++){
    var o = segs[i];
    if(o.a !== a) continue;
    var r = res(o);
    if(r === 'ok' && !o.e) continue;
    h += '<div class="qc-fbi"><b>Einheit ' + o.n + '</b> ' + SYM[r] +
         '<div class="qc-small">Deine Codierung: ' + chip(marks[o.idx] || null, myLabel(o)) +
         ' Musterlösung: ' + chip(o.k, solLabel(o)) +
         (o.alt.length ? '<br>Auch vertretbar: <b>' + esc(altLabel(o)) + '</b>' : '') + '</div>' +
         (o.e ? '<div>' + esc(o.e) + '</div>' : '') + '</div>';
  }
  return h ? '<div class="qc-fb"><b>Musterlösung zu diesem Ausschnitt</b>' + h + '</div>' : '';
}
function scoreHtml(){
  var n = segs.length;
  var hk = 0;
  var hn = 0;
  var sm = 0;
  var sk = 0;
  for(var i = 0; i < n; i++){
    var o = segs[i];
    var m = marks[o.idx] || null;
    if(m === o.k){
      hk++;
      if(SUBS_AN && o.k){ sm++; if(subsel[o.idx] === o.s){ sk++; } }
    } else if(m && altHaupt(o, m)){ hn++; }
  }
  var h = '<div class="qc-score"><b>Hauptkategorie:</b> ' + hk + ' von ' + n + ' Sinneinheiten stimmen mit der Musterlösung überein';
  if(hn > 0){ h += ' (dazu ' + hn + ' vertretbare Alternativen ≈)'; }
  h += '.';
  if(SUBS_AN){ h += '<br><b>Subkategorie:</b> ' + sk + ' von ' + sm + ' Einheiten (gezählt nur dort, wo die Hauptkategorie stimmt und die Musterlösung eine Subkategorie hat).'; }
  return h + '<br>Die Musterlösung ist ein Vorschlag und Anlass zum Vergleich im Plenum.</div>';
}
function stepText(){
  if(phase === 1){ return '<b>Schritt 1' + (SUBS_AN ? ' von 2' : '') + ':</b> Markiere die Sinneinheiten mit dem passenden Stift.'; }
  if(phase === 2){ return '<b>Schritt 2 von 2:</b> Wähle für jede markierte Einheit die passende Subkategorie.'; }
  return '<b>Musterlösung:</b> ✓ stimmt &middot; ≈ vertretbare Alternative (Grenzfall) &middot; ◐ Hauptkategorie stimmt, Subkategorie weicht ab &middot; ✗ weicht ab';
}

function render(){
  var root = document.getElementById('qc-widget');
  if(!root) return;
  var h = '<div class="qc-card"><div class="qc-top">';
  if(phase === 1){
    h += '<div class="qc-bar"><span class="qc-lbl">Markierstift:</span>';
    var keys = Object.keys(KAT);
    for(var k = 0; k < keys.length; k++){
      var c = KAT[keys[k]];
      h += '<button type="button" class="qc-tool' + (tool === keys[k] ? ' active' : '') + '" data-tool="' + keys[k] +
           '" style="background:' + c.farbe + ';border-color:' + c.rand + '">' + esc(c.n) + '</button>';
    }
    h += '<button type="button" class="qc-tool' + (tool === 'none' ? ' active' : '') + '" data-tool="none">Radierer</button></div>';
  }
  h += '<div class="qc-step">' + stepText() + '</div></div>';
  for(var a = 0; a < AUS.length; a++){
    h += '<div class="qc-aus"><div class="qc-h">' + esc(AUS[a].titel) + '</div>';
    var us = AUS[a].units;
    for(var i = 0; i < us.length; i++){ h += wrap(us[i]); }
    if(phase === 3){ h += fbHtml(a); }
    h += '</div>';
  }
  if(phase === 3){ h += scoreHtml(); }
  h += '<div class="qc-actions">';
  if(phase === 1){ h += '<button type="button" class="qc-btn" data-act="next">' + (SUBS_AN ? 'Weiter zu Schritt 2' : 'Auswerten') + '</button>'; }
  if(phase === 2){
    h += '<button type="button" class="qc-btn" data-act="back">Zurück zu Schritt 1</button>';
    h += '<button type="button" class="qc-btn" data-act="check">Auswerten</button>';
  }
  h += '<button type="button" class="qc-btn" data-act="reset">Neu starten</button></div></div>';
  root.innerHTML = h;
}
function onClick(e){
  var el = e.target.closest('[data-tool],[data-seg],[data-act]');
  if(!el) return;
  var root = document.getElementById('qc-widget');
  if(el.hasAttribute('data-tool')){
    tool = el.getAttribute('data-tool');
    var bs = root.querySelectorAll('[data-tool]');
    for(var b = 0; b < bs.length; b++){ bs[b].classList.toggle('active', bs[b].getAttribute('data-tool') === tool); }
  } else if(el.hasAttribute('data-seg')){
    if(phase !== 1) return;
    var i = parseInt(el.getAttribute('data-seg'), 10);
    var old = marks[i] || null;
    marks[i] = (tool === 'none' || old === tool) ? null : tool;
    delete subsel[i];
    root.querySelector('[data-w="' + i + '"]').innerHTML = unitHtml(segs[i]) + extras(segs[i]);
  } else {
    var act = el.getAttribute('data-act');
    if(act === 'next'){ phase = SUBS_AN ? 2 : 3; }
    if(act === 'back'){ phase = 1; }
    if(act === 'check'){ phase = 3; }
    if(act === 'reset'){ marks = {}; subsel = {}; phase = 1; }
    render();
    root.scrollIntoView();
  }
}
function onChange(e){
  if(!e.target.hasAttribute('data-sub')) return;
  var i = parseInt(e.target.getAttribute('data-sub'), 10);
  if(e.target.value){ subsel[i] = e.target.value; } else { delete subsel[i]; }
}
function init(tries){
  var root = document.getElementById('qc-widget');
  if(!root){
    if(tries > 0){ setTimeout(function(){ init(tries - 1); }, 250); }
    return;
  }
  if(root.getAttribute('data-ready')) return;
  root.setAttribute('data-ready', '1');
  prep();
  addCss();
  render();
  root.addEventListener('click', onClick);
  root.addEventListener('change', onChange);
}
init(40);
